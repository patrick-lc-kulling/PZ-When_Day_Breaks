    /* Electric stun weapons. The source material says conductive electrical
       weapons are "partially effective at immobilizing" the liquefied, so
       these do almost no damage - the knockdown is the weapon.

       The knockdown itself is forced from Lua (see StunWeapons.lua), not from
       AlwaysKnockdown: every vanilla user of that flag is a sledgehammer-class
       weapon, and doing it in Lua lets the battery charge gate it.

       Battery charge lives in item modData and is mirrored onto the condition
       bar, so these deliberately carry no repairwithtape/repairwithglue tags -
       otherwise duct tape would recharge a battery - and a firearm-grade
       ConditionLowerChanceOneIn so ordinary wear doesn't fight the gauge. */

    item Daybreak_StunGun
    {
        DisplayCategory = Weapon,
        ItemType = base:weapon,
        Weight = 0.4,
        Icon = NightStick,
        WeaponSprite = Nightstick,
        AttachmentType = Nightstick,

        /* Contact range - shorter than anything in vanilla except a bare fist.
           You have to be close enough to touch the thing.

           It jabs like a knife, and the engine takes that literally:
           WeaponType.getWeaponType classifies a weapon by its SwingAnim alone,
           "Stab" means KNIFE, and a knife attacking an unaware zombie from
           behind gets a forced critical hit - the backstab kill.

           The base:nocriticals tag below is what stops it. It is the last
           check CombatManager.pressedAttack makes, it clears the critical hit
           on every path including the forced one, and without a crit the
           knife-kill animation (the only thing that sets KnifeDeath) never
           plays. Vanilla gives the plunger spear the same tag for the same
           reason. Never remove it while SwingAnim is Stab.

           No CloseKillMove either: that is the separate gate for the jaw stab,
           and every vanilla item that sets it is a base:smallblade. */
        Categories = base:smallblunt,
        SubCategory = Stab,
        SwingAnim = Stab,
        MinRange = 0.61,
        MaxRange = 0.85,
        MinAngle = 0.65,
        WeaponLength = 0.2,

    /* One recipe covers the prod and the stun gun: both carry the
       daybreak_usesbattery tag, and tags[] is an OR. The vanilla
       InsertBattery/RemoveBattery recipes can't be reused here - their
       IsEmpty/NotEmpty flags read drainable uses, which a weapon has no such
       thing as. OnTest keeps it off the menu unless there is room to charge. */
    craftRecipe Daybreak_RechargeStunWeapon
    {
        timedAction = MakingElectrical,
        time = 40,
        AllowBatchCraft = false,
        Tags = InHandCraft;CanBeDoneInDark,
        category = Electrical,
        OnTest = DaybreakStunWeapons.canRecharge,
        OnCreate = DaybreakStunWeapons.onRecharge,
        inputs
        {
            item 1 tags[base:daybreak_usesbattery] mode:keep,
            item 1 [Base.Battery] mode:destroy,
        }
        outputs
        {
        }
    }

    craftRecipe Daybreak_RebuildTaserCartridge
    {
        timedAction = MakingElectrical,
        time = 120,
        SkillRequired = Electricity:3,
        xpAward = Electricity:5,
        Tags = AnySurfaceCraft,
        category = Electrical,
        inputs
        {
            item 1 tags[base:screwdriver] mode:keep flags[Prop1],
            item 1 [Base.Daybreak_TaserCartridge_Spent] mode:destroy,
            item 1 [Base.ElectricWire] mode:destroy,
            item 1 [Base.Battery] mode:destroy,
        }
        outputs
        {
            item 1 Base.Daybreak_TaserCartridge,
        }
    }
