       vanilla combat never doubles up on top of it. Tune behaviour there
       (or via the matching Daybreak SCP-001 sandbox options) rather than
       here - the numbers below only control the vanilla shot itself
       (accuracy, range, sound, reload), not how much it hurts or how long it
       drops someone. */
    item Daybreak_Taser

        DisplayCategory = Weapon,

        WorldStaticModel = Pistol2,
        Weight = 1.1,
        Categories = Firearm,
        SubCategory = Handgun,
        ConditionMax = 20,
        ConditionLowerChanceOneIn = 80,
        RequiresEquippedBothHands = FALSE,

        MinDamage = 0.01,
        MaxDamage = 0.02,
        DoorDamage = 1,
        TreeDamage = 0,
        SplatBloodOnNoDeath = FALSE,

        MaxRange = 6,
        MinAngle = 0.85,
        HitChance = 75,
        AimingPerkHitChanceModifier = 10,

        AimingPerkMinAngleModifier = 0.05,
        CriticalChance = 0,

        KnockdownMod = 0.0,

        SoundVolume = 40,

        ClipSize = 1,
